\begin{tabular}{>{\raggedleft}p{10cm} >{$}c<{$}}
    \text{Equació de continuïtat:} & \frac{\partial \rho_1}{\partial t} + (\bm{v}_1 \cdot \bm{\nabla}) \rho_0 + \rho_0 (\bm{\nabla} \cdot \bm{v}_1) = 0 \\[0.8em]
    \text{Equacions de moviment:} & \rho_0 \frac{\partial \bm{v}_1}{\partial t} = - \bm{\nabla} p_1 + (\bm{\nabla} \bm{\times} \bm{B}_1) \bm{\times} \bm{B}_0 / \mu \\[0.8em]
    \shortstack[c]{Equació d'energia per\\canvis d'estat adiabàtics:} & \frac{\partial p_1}{\partial t} + (\bm{v}_1 \cdot \bm{\nabla}) p_0 - c_s^2 \left( \frac{\partial \rho_1}{\partial t} + (\bm{v}_1 \cdot \bm{\nabla}) \rho_0 \right) = 0 \\[0.8em]
    \text{Equacions d'inducció:} & \frac{\partial \bm{B}_1}{\partial t} = \bm{\nabla} \bm{\times} (\bm{v}_1 \bm{\times} \bm{B}_0) \\[0.8em]
    \text{Llei de Gauss magnètica:} & \bm{\nabla} \cdot \bm{B}_1 = 0 \\[0.8em]
    \text{Velocitat del so:} & c_s^2 = \frac{p_1}{\rho_1} = \frac{\gamma p_0}{\rho_0} = \frac{\gamma k_B T_0}{m}
\end{tabular}
